\begin{frame}{ReLU as a gate}
	\MLPSlideFont\setlength{\parskip}{4pt}\setlength{\abovedisplayskip}{6pt}\setlength{\belowdisplayskip}{6pt}
	\begin{columns}[c,onlytextwidth]
		\begin{column}{0.54\textwidth}
			ReLU multiplies its input by a \textbf{gate} that depends on the input itself:
			\[
				\operatorname{ReLU}(s)=s\,g(s),\qquad
				g(s)=\begin{cases}0 & s<0,\\1 & s\geq0.\end{cases}
			\]
			Changing the gate gives other activations:
			\begin{itemize}
				\item \textbf{LeakyReLU}: $g(s)=\alpha$ for $s<0$ (e.g., $\alpha=0.01$). \item \textbf{PReLU}, same as LeakyReLU but $\alpha$ is learned (per-unit).
				\item \textbf{Smooth gates}: any smooth non-decreasing $0<g(s)<1$, as shown on the right.
			\end{itemize}
		\end{column}

\begin{frame}{Smooth gates: SiLU and GELU}
	\MLPSlideFont\setlength{\parskip}{4pt}\setlength{\abovedisplayskip}{6pt}\setlength{\belowdisplayskip}{6pt}
	\begin{columns}[c,onlytextwidth]
		\begin{column}{0.54\textwidth}

			\textbf{SiLU} ({Sigmoid-weighted linear unit}):
			\[
				\operatorname{SiLU}(s)=s\,\sigma(s),\qquad
				\sigma(s)=\frac{1}{1+e^{-s}}.
			\]
			\vspace{0.5em}
			\textbf{GELU} ({Gaussian error linear unit}):
			\[
				\operatorname{GELU}(s)=s\,\Phi(s).
			\]
			where $\Phi$ is the standard Gaussian CDF.

\begin{frame}{Gating with a second projection: GLU and SwiGLU}
	\MLPSlideFont
	\setlength{\parskip}{3pt}
	\setlength{\abovedisplayskip}{5pt}
	\setlength{\belowdisplayskip}{5pt}
	In ReLU and SiLU, each unit gates its own preactivation. In a
	\textbf{gated linear unit} (GLU), the gate comes from a second projection
	of the input:

	\begin{center}
		\input{../../shared/assets/images/lecture_4/swiglu.tex}
	\end{center}
	\[
		\vc{y}=\vc{W}_o\left[(\vc{W}_u\vc{x})\odot\phi(\vc{W}_g\vc{x})\right].
	\]
	With $\phi=\sigma$, the gate lies in $[0,1]$, as in the original GLU.
	Replacing it with SiLU gives \textbf{SwiGLU}, used in Llama and many recent
	language models: its multiplier can also be negative or exceed one.

		{For $\vc{x}\in\mathbb{R}^{D}$,
			$\vc{W}_u,\vc{W}_g\in\mathbb{R}^{M\times D}$ and $\vc{W}_o\in\mathbb{R}^{D\times M}$.
			Biases are omitted.\par}
	\vfill
	{\scriptsize\color{drawgray}
		\href{https://arxiv.org/abs/2002.05202}{Shazeer (2020)};
		\href{https://arxiv.org/abs/2302.13971}{Touvron et al., LLaMA (2023)}.\par}
\end{frame}

\begin{frame}{Mixtures of experts}
	\MLPSlideFont\setlength{\parskip}{3pt}\setlength{\abovedisplayskip}{5pt}\setlength{\belowdisplayskip}{5pt}
	\begin{columns}[c,onlytextwidth]
		\begin{column}{0.56\textwidth}
			A gate can also choose between whole blocks. A \myalert{mixture of experts} (MoE) combines $E$ \textbf{experts} $f_1,\ldots,f_E$ with the same output shape, weighted by a \textbf{router}:
			\[
				\vc{y}=\sum_{e=1}^{E}a_e(\vc{x})\,f_e(\vc{x}),
				\qquad
				\vc{a}(\vc{x})=\operatorname{softmax}(\vc{W}_r\vc{x}).
			\]
		\end{column}
		\begin{column}{0.40\textwidth}
			\centering
			\scalebox{\MLPMoEScale}{\input{../../shared/assets/images/lecture_4/moe.tex}}
		\end{column}
	\end{columns}
	\medskip
	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.49\textwidth}
			\begin{tcolorbox}[nndsblue,courseheight={mlp-moe}{0.20\textheight},boxsep=1mm,top=1mm,bottom=1mm]
				\RaggedRight\textbf{\color{darkcerulean}Dense MoE}\par
				All $E$ experts run; the router reweights them.
			\end{tcolorbox}
		\end{column}
		\begin{column}{0.49\textwidth}
			\begin{tcolorbox}[nndsgreen,courseheight={mlp-moe}{0.20\textheight},boxsep=1mm,top=1mm,bottom=1mm]
				\RaggedRight\textbf{\color{darkgreen}Sparse MoE}\par
				Only the $k$ top-scoring experts run: computation grows with $k$, not $E$.
			\end{tcolorbox}
		\end{column}
	\end{columns}
	Sparse MoEs are common in recent language models. Training must keep the experts balanced, or the router may send most inputs to a few of them.
	\vfill
	{\scriptsize\color{drawgray}\href{https://arxiv.org/abs/1701.06538}{Shazeer et al. (2017)}; \href{https://arxiv.org/abs/2101.03961}{Fedus et al. (2022)}; \href{https://arxiv.org/abs/2401.04088}{Jiang et al. (2024)}.\par}
\end{frame}
