\section{Multilayer perceptrons}
\subsection{Basic definitions}

% XOR and fixed features now close \MLPLinearReference{}.
\begin{frame}{The multilayer perceptron}
	\setlength{\abovedisplayskip}{5pt}\setlength{\belowdisplayskip}{5pt}
	A \textbf{fully-connected neural network}, also known as a \textbf{multilayer perceptron (MLP)}, learns an intermediate representation of the input:

	\vspace{0.2em}
	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.49\textwidth}
			\begin{tcolorbox}[nndsblue,courseheight={mlp-definition}{0.42\textheight}]
				\textbf{\color{darkcerulean}Hidden representation}
				\[
					{\color{darkcerulean}\vc{h}} = \phi(\vc{W}\vc{x}+\vc{b})
				\]
				\small\RaggedRight $\phi$ is an elementwise \textbf{activation function}.
			\end{tcolorbox}
		\end{column}
		\begin{column}{0.49\textwidth}
			\begin{tcolorbox}[nndsgreen,courseheight={mlp-definition}{0.42\textheight}]
				\textbf{\color{darkgreen}Linear output}
				\[
					{\color{darkgreen}f(\vc{x})} = \vc{w}^{\top}{\color{darkcerulean}\vc{h}}+c
				\]
				\small\RaggedRight A linear model applied to the learned features $\vc{h}$.
			\end{tcolorbox}
		\end{column}
	\end{columns}

	\vspace{0.5em}
	The parameters $\vc{W},\vc{b},\vc{w},c$ are learned jointly from the data.

	\vfill
	{\scriptsize\color{drawgray}Without the nonlinearity ($\phi(z)=z$), the two affine maps collapse:
		\[
			f(\vc{x})=\vc{w}^{\top}(\vc{W}\vc{x}+\vc{b})+c
			=(\vc{w}^{\top}\vc{W})\vc{x}+(\vc{w}^{\top}\vc{b}+c).
		\]}
\end{frame}

\begin{frame}{Embedding the data}
	\begin{columns}[c,onlytextwidth]
		\begin{column}{0.36\textwidth}
			The hidden representation provides new coordinates for the input, in which a linear output layer can distinguish the classes.
		\end{column}
		\begin{column}{0.60\textwidth}
			\centering
			\includegraphics[height=0.80\textheight]{images/lecture_3/embedding.pdf}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Adding hidden layers}
	\small
	\setlength{\abovedisplayskip}{4pt}\setlength{\belowdisplayskip}{4pt}
	Nothing prevents us from stacking $L$ \textbf{hidden} layers before the linear output:

	\begin{center}
		\input{../../shared/assets/images/lecture_4/deep-mlp.tex}
	\end{center}

	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.49\textwidth}
			\begin{tcolorbox}[nndsblue,courseheight={mlp-hidden}{0.32\textheight},fontupper=\small]
				\textbf{\color{darkcerulean}Hidden layers}
				\[
					{\color{darkcerulean}\vc{h}^{(\ell)}}=\phi_{\ell}\!\left(\vc{W}^{(\ell)}\vc{h}^{(\ell-1)}+\vc{b}^{(\ell)}\right)
				\]
				\small\raggedright Each layer $\ell=1,\ldots,L$ has its own weights and activation; $\vc{h}^{(0)}=\vc{x}$.
			\end{tcolorbox}
		\end{column}
		\begin{column}{0.49\textwidth}
			\begin{tcolorbox}[nndsgreen,courseheight={mlp-hidden}{0.32\textheight},fontupper=\small]
				\textbf{\color{darkgreen}Linear output}
				\[
					{\color{darkgreen}f_{\boldsymbol{\theta}}(\vc{x})}=\vc{w}^{\top}\vc{h}^{(L)}+c
				\]
				\small\raggedright The same linear model as before, applied to the last representation.
			\end{tcolorbox}
		\end{column}
	\end{columns}

	\smallskip
	{\RaggedRight
	With widths $D_0,\ldots,D_L$ ($D_0$ is the input dimension), $\vc{W}^{(\ell)}\in\mathbb{R}^{D_\ell\times D_{\ell-1}}$ and $\vc{b}^{(\ell)}\in\mathbb{R}^{D_\ell}$: layer $\ell$ has $D_\ell(D_{\ell-1}+1)$ parameters, and the output adds $D_L+1$.\par
	The parameters are $\boldsymbol{\theta}=\{\vc{W}^{(\ell)},\vc{b}^{(\ell)}\}_{\ell=1}^{L}\cup\{\vc{w},c\}$.\par}
\end{frame}

\begin{frame}{Training the network}
	\small
	\setlength{\abovedisplayskip}{6pt}\setlength{\belowdisplayskip}{6pt}
	\begin{tcolorbox}[nndsgreen]
		Given a dataset $\mathcal{S}=\{(\vc{x}_i,y_i)\}_{i=1}^{N}$, we minimize:
		\[
			\boldsymbol{\theta}^{*}=\arg\min_{\boldsymbol{\theta}}\frac{1}{N}\sum_{i=1}^{N}
			\ell\!\left(f_{\boldsymbol{\theta}}(\vc{x}_i),y_i\right).
		\]
	\end{tcolorbox}
	We use the losses introduced in \MLPLinearReference{}, with the appropriate output for regression or classification (i.e., adding a softmax or sigmoid for multi-class or binary classification, respectively). In order to compute gradients we can use \myalert{automatic differentiation}.
\end{frame}

\begin{frame}{Universal approximation}
	\setlength{\abovedisplayskip}{6pt}\setlength{\belowdisplayskip}{6pt}
	\small
	\begin{tcolorbox}[nndsblue,fontupper=\small]
		Let $g:K\to\mathbb{R}$ be continuous on a compact set $K\subset\mathbb{R}^{d}$, and let $\phi$ be continuous and non-polynomial.

		For any $\varepsilon>0$, there exists an MLP with \textbf{one hidden layer}, sufficiently large width, and biases such that:
		\[
			\sup_{\vc{x}\in K}\left|g(\vc{x})-f_{\boldsymbol{\theta}}(\vc{x})\right|<\varepsilon.
		\]
	\end{tcolorbox}
	This is an \textbf{existence result}. It does not tell us how much width or data we need, or whether training will find suitable parameters.

	A single hidden layer can be sufficient for approximation, but it need not be an efficient choice. We will see shortly how depth can help.
	\SlideSource{Leshno, Lin, Pinkus and Schocken (1993), \href{https://doi.org/10.1016/S0893-6080(05)80131-5}{Neural Networks, 6(6), 861--867}.}
\end{frame}

\begin{frame}{Some terminology}
	\small
	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.49\textwidth}
			\begin{tcolorbox}[nndsblue,courseheight={mlp-terminology}{0.32\textheight}]
				\textbf{\color{darkcerulean}Representation / embedding}

				The feature vector $\vc{h}^{(\ell)}$ computed at an intermediate layer.
			\end{tcolorbox}
		\end{column}
		\begin{column}{0.49\textwidth}
			\begin{tcolorbox}[nndsblue,courseheight={mlp-terminology}{0.32\textheight}]
				\textbf{\color{darkcerulean}Width and depth}

				Width: $\dim\vc{h}^{(\ell)}$. Depth: $L$ hidden layers ($L+1$ with the output).
			\end{tcolorbox}
		\end{column}
	\end{columns}
	\par\vspace{0.6em}
	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.49\textwidth}
			\begin{tcolorbox}[nndsgreen,courseheight={mlp-terminology}{0.32\textheight}]
				\textbf{\color{darkgreen}Parameters: learned}

				The weights and biases $\boldsymbol{\theta}$ learned from the data.
			\end{tcolorbox}
		\end{column}
		\begin{column}{0.49\textwidth}
			\begin{tcolorbox}[nndsorange,courseheight={mlp-terminology}{0.32\textheight}]
				\textbf{\color{draworange}Hyperparameters: chosen}

				Layer count, widths, activation functions, and learning rate.
			\end{tcolorbox}
		\end{column}
	\end{columns}
	\vspace{0.5em}
	Choosing hyperparameters is called \textbf{model selection} / \textbf{hyperparameter optimization}.
\end{frame}

\begin{frame}{The rectified linear unit (ReLU)}
	\begin{columns}[c,onlytextwidth]
		\begin{column}{0.55\textwidth}
			The \myalert{rectified linear unit} is defined as:
			\[
				{\color{darkcerulean}\phi(s)=\max(0,s)}.
			\]
			In words: negative inputs become zero, positive inputs pass through unchanged.
		\end{column}
		\begin{column}{0.41\textwidth}
			\centering
			\begin{tikzpicture}[x=0.62cm,y=0.62cm]
				\draw[->,gray] (-3.4,0)--(3.5,0) node[right] {$s$};
				\draw[->,gray] (0,-0.5)--(0,3.5) node[above] {$\phi(s)$};
				\foreach \x in {-3,-2,-1,1,2,3}
				\draw[gray] (\x,0.07)--(\x,-0.07) node[below,font=\scriptsize] {$\x$};
				\foreach \y in {1,2,3}
				\draw[gray] (0.07,\y)--(-0.07,\y) node[left,font=\scriptsize] {$\y$};
				\draw[darkcerulean,very thick] (-3,0)--(0,0)--(3,3);
			\end{tikzpicture}
		\end{column}
	\end{columns}
	\vspace{1.2em}
	\small ReLU is a good default choice for MLPs: we will explain more in the automatic differentiation lecture, when we study how gradients propagate through the network.
\end{frame}

\begin{frame}{One network, many affine maps}
  \small
  \setlength{\abovedisplayskip}{6pt}\setlength{\belowdisplayskip}{6pt}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.56\textwidth}
      Each ReLU either passes its input or sets it to zero:
      \[
        \vc{h}=\vc{D}(\vc{x})(\vc{W}\vc{x}+\vc{b}),
        \qquad D_{ii}(\vc{x})\in\{0,1\}.
      \]
      The diagonal entries of $\vc{D}$ indicate which units are active. With an affine output,
      \begin{align*}
        f(\vc{x})&=\vc{w}^{\top}\vc{D}(\vc{x})(\vc{W}\vc{x}+\vc{b})+c\\
                 &=\vc{a}(\vc{x})^{\top}\vc{x}+\beta(\vc{x}).
      \end{align*}
      Within each region, the gates stay fixed: the network is \myalert{exactly affine}.

      \medskip
      This holds at any depth. Training learns both the regions and their affine maps.
    \end{column}
    \begin{column}{0.41\textwidth}
      \centering
      $f(\vc{x})=\operatorname{ReLU}(x_1)+\operatorname{ReLU}(x_2-x_1)$\par
      \smallskip
      \input{../../shared/assets/images/lecture_4/relu/regions.tex}
    \end{column}
  \end{columns}
  \SlideSource{For ReLU hidden layers, before any output sigmoid or softmax. \href{https://arxiv.org/abs/1402.1869}{Mont\'ufar et al. (2014)}.}
\end{frame}

\begin{frame}{Depth through composition}
  \small
  \setlength{\abovedisplayskip}{5pt}\setlength{\belowdisplayskip}{5pt}
  Three ReLUs make a triangular function. We apply the same function repeatedly:
  \[
    T(x)=2\operatorname{ReLU}(x)-4\operatorname{ReLU}\!\left(x-\tfrac12\right)
         +2\operatorname{ReLU}(x-1),\qquad x\in[0,1].
  \]
  \begin{center}
    \input{../../shared/assets/images/lecture_4/relu/folding.tex}
  \end{center}
  Each composition splits every piece in two: $k$ blocks give $2^k$ pieces using $3k$ ReLU units.

  \smallskip
  A single hidden layer with $m$ ReLUs has at most $m+1$ pieces in one dimension.
  \myalert{Depth can make a representation much more compact.}

  \SlideSource{An expressivity example, not a guarantee about training or generalization. \href{https://proceedings.mlr.press/v49/telgarsky16.html}{Telgarsky (2016)}.}
\end{frame}

\begin{frame}{Sharing parameters and looping a block}
	\small
	\setlength{\abovedisplayskip}{4pt}\setlength{\belowdisplayskip}{4pt}
	We can use different parameters at each layer, or apply the same block
	repeatedly. The latter gives a \myalert{looped model}.

	\smallskip
	\begin{columns}[c,onlytextwidth]
		\begin{column}{0.60\textwidth}
			\textbf{\color{darkcerulean}Separate parameters}\par\smallskip
			\centering\input{../../shared/assets/images/lecture_4/untied.tex}\par\smallskip
			\raggedright\textbf{\color{darkcerulean}Shared parameters}\par\smallskip
			\centering\input{../../shared/assets/images/lecture_4/shared.tex}
		\end{column}
		\begin{column}{0.37\textwidth}
			\centering
			\input{../../shared/assets/images/lecture_4/loop.tex}
		\end{column}
	\end{columns}

	\medskip
	\begin{tcolorbox}[nndsgreen,boxsep=1mm,top=1mm,bottom=1mm,fontupper=\small]
		The shared block computes $\vc{h}^{(\ell)}=f_{\boldsymbol{\theta}}(\vc{h}^{(\ell-1)})$, with matching input and output shapes. A block with $P$ parameters needs $P$ instead of $LP$, but still $L$ evaluations.
	\end{tcolorbox}

	We will meet looped blocks again with recurrent networks.
	\SlideSource{\href{https://arxiv.org/abs/1807.03819}{Dehghani et al., Universal Transformers (2019)}.}
\end{frame}

\begin{frame}{Shared MLPs on structured data}
	\small
	\setlength{\abovedisplayskip}{4pt}\setlength{\belowdisplayskip}{4pt}
	We can also share an MLP across the positions of \MLPSequenceDescription{},
	applying it to each position independently:

	\medskip
	\begin{columns}[c,onlytextwidth]
		\begin{column}{0.50\textwidth}
			\[
				\MLPPositionMapping{}
			\]
			The number of parameters does not depend on the sequence length,
			and the same holds for the pixels of an image.
			Transformers use these \textbf{positionwise MLPs} in every block.
		\end{column}
		\begin{column}{0.46\textwidth}
			\centering
			\MLPPositionDiagram{}
		\end{column}
	\end{columns}

	\medskip
	\begin{tcolorbox}[nndsorange,boxsep=1mm,top=1mm,bottom=1mm,fontupper=\small]
		A positionwise MLP mixes the features within a position, but no information moves between positions.
	\end{tcolorbox}

	To exchange information, we need another operation: convolutions over
	neighborhoods (the next lecture), attention, or a recurrent state.
\end{frame}
