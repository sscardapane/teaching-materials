\section{Derivatives}
\subsection{Derivatives and gradients}

\begin{frame}{Empirical risk minimization}

	For a dataset $\left\{(\vc{x}_i, y_i)\right\}_{i=1}^N$ (more on this next lecture), let $\boldsymbol{\theta}$ denote the trainable parameters of a model $f_{\boldsymbol{\theta}}(\vc{x})$ (e.g., for a linear layer $\boldsymbol{\theta}=\{\vc{W},\vc{b}\}$) and define:
	%
	\begin{equation}
		\mathcal{L}(\boldsymbol{\theta}) =
		\frac{1}{N}\sum_{i=1}^{N}
		\ell\Bigl(f_{\boldsymbol{\theta}}(\vc{x}_i),y_i\Bigr)\,,
	\end{equation}
	%
	where $\ell$ is some per-example loss. Training is the optimization problem:
	%
	\begin{equation*}
		\boldsymbol{\theta}^* = \argmin_{\boldsymbol{\theta}}\mathcal{L}(\boldsymbol{\theta})\,.
	\end{equation*}

	How does $\mathcal{L}$ change when we perturb $\boldsymbol{\theta}$? This is what derivatives tell us.

\end{frame}

\begin{frame}{Derivative}

	Most of this course is based on the notion of derivative.

	\begin{tcolorbox}[nndsblue]
		For $f:\mathbb{R}\rightarrow\mathbb{R}$, the \myalert{derivative} is:
		\[
			\partial f(x) = \frac{\partial}{\partial x}f(x) = \pr{f}(x) = \lim_{h \rightarrow 0} \frac{f(x+h) - f(x)}{h} \,.
		\]
		Even for a continuous function, $\partial f(x)$ might not be defined everywhere.
	\end{tcolorbox}

	Informally, the derivative expresses the rate of change of $f$ around an infinitesimal displacement from $x$, or the slope of the line tangent to $f(x)$.

\end{frame}

\begin{whiteframe}{Visualizing derivatives in the 1D case}

	\begin{figure}
		\centering\includegraphics[width=8cm,keepaspectratio]{images/lecture_2/gradient_info.pdf} %
		\caption{1D function ($f(x) = x^2 - 1.5x$), showing the derivative at two different locations.}
	\end{figure}

\end{whiteframe}

\begin{frame}{Properties of derivatives}

	Derivatives possess a number of properties, most notably:

	\bigskip
	\begin{center}{\renewcommand{\arraystretch}{2.2}
		\begin{tabular}{@{}l@{\hspace{2.5em}}l@{}}
			\textbf{Linearity}    &
			$\partial \Big[{\color{darkred}f(x)} + {\color{darkgreen}g(x)}\Big] = {\color{darkred}\pr{f}(x)} + {\color{darkgreen}g'(x)}$                                                     \\
			\textbf{Product rule} &
			$\partial\Big[ {\color{darkred}f(x)}{\color{darkgreen}g(x)}\Big] = {\color{darkred}\pr{f}(x)}{\color{darkgreen}g(x)} + {\color{darkred}f(x)}{\color{darkgreen}g'(x)}$         \\
			\textbf{Chain rule}   &
			$\partial \Big[{\color{darkred}f(}{\color{darkgreen}g(x)}{\color{darkred})}\Big] = {\color{darkred}\pr{f}(}{\color{darkgreen}g(x)}{\color{darkred})}{\color{darkgreen}g'(x)}$
		\end{tabular}}\end{center}

\end{frame}

\begin{frame}{Gradient}

	For $f:\mathbb{R}^D\rightarrow\mathbb{R}$, the \myalert{gradient} $\nabla f(\vc{x})$ is a $D$-dimensional vector defined as:%
	\begin{equation}
		\idx{\nabla f(\vc{x})}{i} = \frac{\partial f(\vc{x})}{\partial x_i} = \lim_{h \rightarrow 0} \frac{f(\vc{x} + h\vc{e}_i) - f(\vc{x})}{h} \,,
	\end{equation}
	%
	where $\vc{e}_i$ is the $i$th standard basis vector:
	%
	\begin{equation*}
		\idx{\vc{e}_i}{j} = \begin{cases} 1 & \text{ if } i=j \\ 0 & \text{otherwise} \end{cases}
	\end{equation*}
	%
	Each entry is the derivative along one coordinate, keeping the others fixed.

\end{frame}

\begin{frame}{Directional derivative}

	\begin{tcolorbox}[nndsblue]
		More generally, the \myalert{directional derivative} of $f:\mathbb{R}^D\rightarrow\mathbb{R}$ in the direction $\vc{u}$ is:
		\[
			\mathrm{D}_{\vc{u}}f(\vc{x}) = \lim_{h \rightarrow 0} \frac{f(\vc{x} + h\vc{u}) - f(\vc{x})}{h} \,.
		\]
	\end{tcolorbox}
	%
	It is easy to prove that:
	%
	\begin{equation}
		\mathrm{D}_{\vc{u}}f(\vc{x}) = \langle \nabla f(\vc{x}), \vc{u} \rangle \,.
	\end{equation}
	%
	A partial derivative is a directional derivative in the direction of a standard basis vector.

\end{frame}

\begin{frame}{Gradients and Jacobians}

	Everything extends to vector-valued functions $f:\mathbb{R}^D\rightarrow\mathbb{R}^H$, $\vc{y} = f(\vc{x})$:
	%
	\begin{tcolorbox}[nndsblue]
		The \myalert{Jacobian} $\partial f(\vc{x})\sim(H,D)$ of $f$ is the matrix:
		\[
			\partial f(\vc{x}) = \begin{pmatrix}
				\frac{\partial y_1}{\partial x_1} & \dots  & \frac{\partial y_1}{\partial x_D} \\
				\vdots                            & \ddots & \vdots                            \\
				\frac{\partial y_H}{\partial x_1} & \dots  & \frac{\partial y_H}{\partial x_D} \\
			\end{pmatrix} \,.
		\]
	\end{tcolorbox}
	%
	For $H=1$, its single row is the gradient; for $D=H=1$, we recover the standard derivative.

\end{frame}

\begin{frame}{Examples of matrix derivatives}

	Three derivatives we will use often:

	\bigskip
	\begin{center}{\renewcommand{\arraystretch}{2.4}
		\begin{tabular}{@{}l@{\hspace{2em}}l@{\hspace{2em}}l@{}}
			Inner product          & $f:\mathbb{R}^D\rightarrow\mathbb{R}$     & $\nabla_{\vc{x}} \langle \vc{x}, \vc{y} \rangle = \vc{y}$ \\
			Linear map             & $f:\mathbb{R}^D\rightarrow\mathbb{R}^H$   & $\partial_{\vc{x}}\, \vc{A}\vc{x} = \vc{A}$                \\
			Squared Euclidean norm & $f:\mathbb{R}^D\rightarrow\mathbb{R}$     & $\nabla \lVert \vc{x} \rVert^2 = 2\vc{x}$
		\end{tabular}}\end{center}

	\SlideSource{See the \href{https://www.math.uwaterloo.ca/~hwolkowi/matrixcookbook.pdf}{Matrix Cookbook} for reference.}

\end{frame}

\begin{frame}{First-order approximation}
	\setlength{\abovedisplayskip}{4pt}
	\setlength{\belowdisplayskip}{4pt}

	\begin{tcolorbox}[nndsblue]
		For $f:\mathbb{R}\rightarrow\mathbb{R}$ and a small displacement $h$, the best \textbf{linear approximation} of $f$ around $x$ is (\myalert{Taylor's theorem}):
		\[
			f(x+h)\approx f(x)+\partial f(x)\,h\,.
		\]
		For $f:\mathbb{R}^D\rightarrow\mathbb{R}$ and a displacement $\vc{h}$:
		\vspace{1.4em}
		\[
			f(\vc{x}+\vc{h})\approx f(\vc{x})+\langle \eqnmarkbox[drawred]{slope}{\nabla f(\vc{x})}, \eqnmarkbox[drawgreen]{disp}{\vc{h}} \rangle\,.
		\]
		\annotate[yshift=0.8em]{above,left}{slope}{Slope}
		\annotate[yshift=0.8em]{above,right}{disp}{Displacement}
	\end{tcolorbox}

	Better approximations use higher-order derivatives, but this is enough for building effective optimization algorithms.

\end{frame}

\section{Optimization}
\subsection{Gradient descent and convexity}

\begin{frame}{Optimizing a function}

	We can use gradients to solve generic problems of the form:
	%
	\begin{equation}
		\boldsymbol{\theta}^* = \argmin_{\boldsymbol{\theta}} \mathcal{L}(\boldsymbol{\theta})
	\end{equation}
	%
	Note that maximizing/minimizing are equivalent in the sense that:
	%
	\begin{equation}
		\boldsymbol{\theta}^* = \argmax_{\boldsymbol{\theta}} \mathcal{L}(\boldsymbol{\theta}) = \argmin_{\boldsymbol{\theta}} {\color{darkred} -} \mathcal{L}(\boldsymbol{\theta})
	\end{equation}

\end{frame}

\begin{frame}{Minima and stationary points}

	\begin{tcolorbox}[nndsblue]
		A point $\boldsymbol{\theta}$ such that $\mathcal{L}(\boldsymbol{\theta}) \le \mathcal{L}(\boldsymbol{\theta}')$ for all $\boldsymbol{\theta}'$ is a \myalert{global minimum}. It is a \myalert{local minimum} if this holds only nearby:
		\[
			\mathcal{L}(\boldsymbol{\theta}) \le \mathcal{L}(\boldsymbol{\theta}') \quad \text{for all } \boldsymbol{\theta}' \text{ with } \lVert \boldsymbol{\theta}' - \boldsymbol{\theta} \rVert < \varepsilon\,,
		\]
		for some $\varepsilon > 0$.
	\end{tcolorbox}

	\begin{tcolorbox}[nndsblue]
		If $\nabla \mathcal{L}(\boldsymbol{\theta}) = \vc{0}$, $\boldsymbol{\theta}$ is a \myalert{stationary point}. Stationary points can be minima, maxima, or \myalert{saddle points}.
	\end{tcolorbox}

	First-order optimization algorithms return stationary points, in practice local minima or saddle points.

\end{frame}

\begin{whiteframe}{Types of stationary points}

	\begin{figure}
		\centering\includegraphics[width=9.5cm,keepaspectratio]{images/lecture_2/stationary_points.pdf} %
		\vspace{-1em}\caption{Stationary points can be \myalert{local} or
			\myalert{global} extrema, or \myalert{saddle points}; here, the saddle
			is a stationary inflection.}
	\end{figure}

\end{whiteframe}

\begin{frame}{Iterative optimization}

	\begin{tcolorbox}[nndsblue]
		Given a randomly initialized $\boldsymbol{\theta}_0$, an \myalert{iterative optimization algorithm} has the form:
		\vspace{1.2em}
		\[
			\boldsymbol{\theta}_t = \eqnmarkbox[drawred]{prev}{\boldsymbol{\theta}_{t-1}} + \eqnmarkbox[drawblue]{step}{\eta_t\vc{p}_t}\,.
		\]
		\annotate[yshift=0.8em]{above,left}{prev}{Guess at iteration $t-1$}
		\annotate[yshift=0.8em]{above,right}{step}{Displacement at iteration $t$}
	\end{tcolorbox}

	$\vc{p}_t$ is a \myalert{descent direction} if $\mathcal{L}(\boldsymbol{\theta}_t) < \mathcal{L}(\boldsymbol{\theta}_{t-1})$ for a sufficiently small $\eta_t>0$, called the \myalert{step size} or \myalert{learning rate}.

	To implement this algorithm, we need a way to choose a descent direction (and a step size) at every iteration.

\end{frame}

\begin{frame}[fragile]{Descent direction}
	\setlength{\abovedisplayskip}{4pt}
	\setlength{\belowdisplayskip}{4pt}

	We restrict to unit directions, $\lVert\vc{p}_t\rVert=1$, and recall that
	$\langle\vc{a},\vc{b}\rangle=\lVert\vc{a}\rVert\lVert\vc{b}\rVert\cos(\alpha)$,
	where $\alpha$ is the angle between the two vectors.

	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.62\textwidth}
			The rate of change along $\vc{p}_t$ is:
			\begin{align*}
				\mathrm{D}_{\vc{p}_t}\mathcal{L}(\boldsymbol{\theta}_{t-1})
				 & =
				\left\langle
				{\color{darkred}\nabla\mathcal{L}(\boldsymbol{\theta}_{t-1})},
				{\color{darkpastelblue}\vc{p}_t}
				\right\rangle
				\\
				 & =
				\left\lVert\nabla\mathcal{L}(\boldsymbol{\theta}_{t-1})\right\rVert
				\underbrace{\lVert\vc{p}_t\rVert}_{=1}
				{\color{draworange}\cos(\alpha)}\,.
			\end{align*}
		\end{column}

		\begin{column}{0.34\textwidth}
			\centering
\begin{tikzpicture}[>=stealth, line cap=round]
				\coordinate (origin) at (0,0);
				\draw[
					->,
					darkred,
					line width=1.3pt
				]
				(origin) -- (1.7,0)
				node[below=3pt, text=darkred]
				{$\nabla\mathcal{L}$};
				\draw[
					->,
					darkpastelblue,
					line width=1.3pt
				]
				(origin) -- (-1.7,0)
				node[below=3pt, text=darkpastelblue]
				{$\vc{p}_t$};
				\draw[
					->,
					draworange,
					line width=0.8pt
				]
				(0.70,0.02) arc[start angle=0,end angle=180,radius=0.70];
				\node[text=draworange] at (0,0.88) {$\alpha=\pi$};
				\fill[drawgray] (origin) circle (1.2pt);
			\end{tikzpicture}

			\medskip
			Minimum at ${\color{draworange}\cos(\alpha)}=-1$:
			\[
				\vc{p}_t\propto-\nabla\mathcal{L}(\boldsymbol{\theta}_{t-1})\,.
			\]
		\end{column}
	\end{columns}

	This is the \myalert{steepest descent direction}. More generally,
	any direction with $\cos(\alpha)<0$ is a descent direction.

\end{frame}

\begin{frame}{Gradient descent}

	The resulting algorithm is called \myalert{gradient descent}.

	\begin{tcolorbox}[nndsgreen]
		Gradient descent (GD) looks for stationary points by iterating:
		\[
			\boldsymbol{\theta}_t = \boldsymbol{\theta}_{t-1} - \eta_t \nabla \mathcal{L}(\boldsymbol{\theta}_{t-1}) \,.
		\]
	\end{tcolorbox}

	This is the basic \textbf{first-order} method. The steepest local direction is not necessarily the \textit{fastest} way to the minimum, and we will see how to accelerate it at the end of this lecture.

	In the interactive part (2c), we see how $\eta_t$ decides between slow, oscillating and diverging runs.

\end{frame}

\begin{frame}{Definition of a convex function}

	\myalert{Convexity} plays a pivotal role in optimization: convex functions are much easier to optimize than non-convex ones.

	\begin{tcolorbox}[nndsblue]
		$f$ is \myalert{convex} if, for any $\vc{x}_1,\vc{x}_2$ and $\lambda \in [0,1]$:

		\begin{columns}[c,onlytextwidth]
			\begin{column}{0.68\textwidth}
				\[
					\textcolor{darkgreen}{f\Bigl((1-\lambda)\vc{x}_1 + \lambda \vc{x}_2\Bigr)}
					\le
					\textcolor{darkred}{(1-\lambda)f(\vc{x}_1) + \lambda f(\vc{x}_2)} \,.
				\]
				The graph lies \textcolor{darkgreen}{below} the
				\textcolor{darkred}{chord} joining any two of its points.
			\end{column}
			\begin{column}{0.29\textwidth}
				\centering
\begin{tikzpicture}[x=1.05cm,y=0.82cm]
					\draw[darkcerulean,very thick,domain=-1.55:1.35,samples=50]
					plot (\x,{0.58*(\x)^2});
					\coordinate (xone) at (-1.35,1.057);
					\coordinate (xtwo) at (1.10,0.702);
					\coordinate (chordpoint) at (0,0.861);
					\coordinate (curvepoint) at (0,0);
					\draw[darkred,very thick] (xone) -- (xtwo);
					\draw[densely dashed,gray] (chordpoint) -- (curvepoint);
					\fill[darkcerulean] (xone) circle (2.1pt)
					node[above left=-1pt] {\scriptsize $\vc{x}_1$};
					\fill[darkcerulean] (xtwo) circle (2.1pt)
					node[above right=-1pt] {\scriptsize $\vc{x}_2$};
					\fill[darkred] (chordpoint) circle (2.1pt);
					\fill[darkgreen] (curvepoint) circle (2.1pt);
				\end{tikzpicture}
			\end{column}
		\end{columns}
	\end{tcolorbox}

	If the inequality is strict for $\vc{x}_1\neq\vc{x}_2$ and $\lambda\in(0,1)$, then $f$ is \myalert{strictly convex}.

\end{frame}

\begin{frame}{What gradient descent can guarantee}
	\setlength{\parskip}{4pt}

	Under standard regularity and step-size assumptions, suppose (S)GD converges to a stationary point $\boldsymbol{\theta}^*$. More structure in $\mathcal{L}$ gives a stronger conclusion:

	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.32\textwidth}
			\begin{tcolorbox}[nndsorange,courseheight={opt-guarantees}{}]
				\textbf{Non-convex}

				Only \textbf{stationarity}: $\boldsymbol{\theta}^*$ may be a local minimum or a saddle point.
			\end{tcolorbox}
		\end{column}
		\begin{column}{0.32\textwidth}
			\begin{tcolorbox}[nndsblue,courseheight={opt-guarantees}{}]
				\textbf{Convex}

				Every stationary point is a \textbf{global minimum}.
			\end{tcolorbox}
		\end{column}
		\begin{column}{0.32\textwidth}
			\begin{tcolorbox}[nndsgreen,courseheight={opt-guarantees}{}]
				\textbf{Strictly convex}

				If a minimum exists, it is \textbf{unique}.
			\end{tcolorbox}
		\end{column}
	\end{columns}

	\medskip
	Neural network losses are non-convex. Globally optimizing a general non-convex objective can be NP-hard, so in practice we settle for good stationary points.

\end{frame}

\subsection{Stochastic gradient descent}

\begin{frame}{Handling large datasets}

	Consider the steps needed for a single iteration of GD on our training problem:

	\begin{enumerate}
		\item Computing the output of the model $f_{\boldsymbol{\theta}}(\vc{x}_i)$ on \textit{all} examples;
		\item Computing the gradient of the loss with respect to the parameters.
	\end{enumerate}

	Both operations scale \textit{linearly} in the number of examples $N$, which is unfeasible when we have $10^5$ examples or more.

	In practice, to train neural networks we use \myalert{stochastic} versions of GD, which approximate the gradient from smaller amounts of data.

\end{frame}

\begin{frame}{Stochastic gradient descent}
	\setlength{\parskip}{4pt}

	The loss is an average over all $N$ examples. We approximate it with a \myalert{mini-batch} $\mathcal{B}_t$ of $B$ examples, sampled at random:
	\[
		\widetilde{\mathcal{L}}_t(\boldsymbol{\theta})
		= {\color{darkred}\frac{1}{B} \sum_{i \in \mathcal{B}_t}}
		\ell(f_{\boldsymbol{\theta}}(\vc{x}_i), y_i)\,.
	\]
	The mini-batch gradient $\nabla\widetilde{\mathcal{L}}_t$ is an unbiased, noisy estimate of the full gradient $\nabla\mathcal{L}$.

	\begin{tcolorbox}[nndsgreen]
		\myalert{Stochastic gradient descent} (SGD) uses $\nabla\widetilde{\mathcal{L}}_t$ in place of $\nabla\mathcal{L}$. One iteration costs $O(B)$, regardless of $N$. With a suitable learning-rate schedule, it still drives the expected gradient norm towards zero.
	\end{tcolorbox}

\end{frame}

\begin{whiteframe}{Full-batch vs. mini-batch trajectories}

	\begin{figure}
		\centering
		\includegraphics[width=0.92\columnwidth]{images/lecture_2/full_vs_minibatch.pdf}
		\caption{Comparison of standard GD (``full-batch'') and SGD:
			mini-batch sampling introduces noise in each update, but the loss
			still decreases on average.}
	\end{figure}

\end{whiteframe}

\begin{whiteframe}{Extracting batches from the data}
	\setlength{\parskip}{3pt}

	Instead of sampling a new batch at every iteration, we usually
	\textbf{shuffle} the dataset, \textbf{split} it into blocks of $B$
	examples processed one after the other, and \textbf{repeat}.

	\begin{center}
		\includegraphics[width=0.72\columnwidth]{images/lecture_2/epoch_batches.pdf}
	\end{center}

	A full pass over the dataset is an \myalert{epoch}. Reading contiguous
	blocks is efficient, and epochs are convenient to define metrics and
	stopping criteria.

\end{whiteframe}

\begin{whiteframe}{Parallelizing GD optimization}
	\setlength{\parskip}{3pt}

	With several GPUs, we replicate the weights and give each GPU its own
	mini-batch. Since the gradient is linear, we can average the per-GPU
	gradients and broadcast the result, which needs synchronization.

	\begin{figure}
		\centering
		\includegraphics[width=0.66\columnwidth]{images/lecture_2/mini_batch.pdf}
	\end{figure}

\end{whiteframe}

\subsection{Momentum and Adam (optional)}

% Figures: shared/assets/images/lecture_2/src/plot_conditioning.py
\begin{whiteframe}{Conditioning}
	\setlength{\parskip}{3pt}

	\begin{columns}[c,onlytextwidth]
		\begin{column}{0.45\textwidth}
			Consider an elongated quadratic:
			\[
				\mathcal{L}(\boldsymbol{\theta})=\tfrac12\left(\theta_1^2+\kappa\,\theta_2^2\right),\quad \kappa=20.
			\]
			The gradient points mostly across the valley. GD oscillates along
			the steep direction and moves slowly along the flat one.
		\end{column}
		\begin{column}{0.52\textwidth}
			\centering
			\includegraphics[width=\linewidth]{images/lecture_2/conditioning.pdf}
		\end{column}
	\end{columns}

	\medskip
	\begin{tcolorbox}[nndsorange]
		To stay stable, the step size must satisfy $\eta<2/\kappa$ for the steep
		direction. The flat direction then shrinks by only $1-\eta$ per step.
	\end{tcolorbox}

	Momentum smooths the path; Adam rescales each coordinate.

\end{whiteframe}

\begin{whiteframe}{Momentum}
	\setlength{\parskip}{3pt}
	\setlength{\abovedisplayskip}{4pt}
	\setlength{\belowdisplayskip}{4pt}

	\begin{columns}[c,onlytextwidth]
		\begin{column}{0.47\textwidth}
			\myalert{Momentum} keeps a running displacement $\vc{u}_t$, with
			$\vc{u}_0=\vc{0}$ and $0\le\lambda<1$:
			\[
				\begin{aligned}
					\vc{u}_t              & = {\color{darkpastelblue}\lambda\vc{u}_{t-1}} {\color{darkred}\;-\;\eta_t\nabla\widetilde{\mathcal{L}}_t(\boldsymbol{\theta}_{t-1})}\,, \\
					\boldsymbol{\theta}_t & = \boldsymbol{\theta}_{t-1}+\vc{u}_t\,.
				\end{aligned}
			\]
			Each step adds the {\color{darkred}new gradient} to a
			{\color{darkpastelblue}fraction of the previous step}.
		\end{column}
		\begin{column}{0.5\textwidth}
			\centering
			\includegraphics[width=\linewidth]{images/lecture_2/momentum_path.pdf}
		\end{column}
	\end{columns}

	\medskip
	\begin{tcolorbox}[nndsgreen]
		Unrolling, $\vc{u}_t$ sums all past gradients with weights $1,\lambda,\lambda^2,\ldots$:
		directions that agree accumulate, oscillations cancel. A common choice is $\lambda=0.9$.
	\end{tcolorbox}

\end{whiteframe}

\begin{frame}{Adam}
	\setlength{\parskip}{3pt}
	\setlength{\abovedisplayskip}{5pt}
	\setlength{\belowdisplayskip}{5pt}

	\myalert{Adam} keeps two averages of $\vc{g}_t=\nabla\widetilde{\mathcal{L}}_t(\boldsymbol{\theta}_{t-1})$:
	\[
		\begin{aligned}
			\vc{m}_t              & = \beta_1\vc{m}_{t-1}+(1-\beta_1)\,\vc{g}_t                                                     &  & {\color{drawgray}\text{average direction}}                \\
			\vc{v}_t              & = \beta_2\vc{v}_{t-1}+(1-\beta_2)\,\vc{g}_t\odot\vc{g}_t                                         &  & {\color{drawgray}\text{average squared size}}             \\
			\boldsymbol{\theta}_t & = \boldsymbol{\theta}_{t-1}-\eta_t\,\frac{\widehat{\vc{m}}_t}{\sqrt{\widehat{\vc{v}}_t}+\varepsilon} &  & {\color{drawgray}\text{per-coordinate step}}
		\end{aligned}
	\]
	Both averages start at zero, so we correct their bias:
	\[
		\widehat{\vc{m}}_t=\frac{\vc{m}_t}{1-\beta_1^t},\qquad
		\widehat{\vc{v}}_t=\frac{\vc{v}_t}{1-\beta_2^t}.
	\]

	\begin{tcolorbox}[nndsgreen]
		Each coordinate is divided by its typical gradient size, so steep and flat
		directions take comparable steps. Defaults: $\beta_1=0.9$, $\beta_2=0.999$.
	\end{tcolorbox}

\end{frame}
