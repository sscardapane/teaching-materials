\section{Introduction}
\subsection{Why fully-connected layers are not enough}

\begin{frame}{Definition of an image}

    An \myalert{image} is a $3$-dimensional tensor $\stackunder{X}{\sss (h,w,c)}$, where:
    %
    \begin{itemize}
        \item $h$ is the \textbf{height} of the image (e.g., 512 pixels).
        \item $w$ is the \textbf{width} of the image (e.g., 1024 pixels).
        \item $c$ is the number of \textbf{channels} (e.g., $3$ channels for a RGB image, $1$ channel for a greyscale image).
    \end{itemize}
    %
    The first two dimensions have a precise \textit{grid} ordering, while the channels do not have a precise ordering (i.e., we can switch RGB to GBR or BRG with no information loss).

\end{frame}

\begin{frame}{An MLP at every pixel}

    As we saw for sequences, we can apply the same MLP at every pixel:
    \[
        \idx{H}{i,j,:} = f_{\boldsymbol{\theta}}(\idx{X}{i,j,:}),
        \qquad H \sim (h,w,c').
    \]

    Each pixel is processed independently. The MLP mixes channels, but does
    not use the neighboring pixels.

    The same parameters are shared across all locations, so the number of
    parameters does not depend on the size of the image.

    How can we also exchange information between pixels?

\end{frame}

\begin{frame}{Vectorizing an image}

    A simple way to process the whole image is to \myalert{vectorize} it,
    apply a fully-connected layer, and reshape the output:
    \[
        \shape{\vc{x}}{(hwc)} = \operatorname{vect}(X),
        \qquad
        H = \operatorname{unvect}\!\left(\phi(\vc{W}\vc{x}+\vc{b})\right).
    \]

    We can choose the output shape to be $(h,w,c')$, so that the layer can
    be composed with other image layers.

    Now every output pixel can depend on the entire image. However, the
    weight matrix has $hwc \cdot hwc'$ parameters. Nothing in this layer
    forces it to use the spatial structure of the image.

\end{frame}

\begin{frame}{An example to keep in mind}
    \small
    \setlength{\parskip}{2pt}
    \setlength{\arraycolsep}{7pt}

    \setlength{\belowdisplayskip}{5pt}

    As a running example to visualize what follows, consider a 1D sequence (think of this as “\textit{4 pixels with a single channel}”):

    $$
        \mathbf{x} = \left[x_1, x_2, x_3, x_4\right]
    $$

    In this case, we do not need any reshaping operations, and the previous layer (with $c^\prime = 1$) can be written as:

    $$
        \begin{bmatrix} h_1 \\ h_2 \\ h_3 \\ h_4 \end{bmatrix}=\begin{bmatrix}W_{11} & W_{12} & W_{13} & W_{14} \\ W_{21} & W_{22} & W_{23} & W_{24} \\ W_{31} & W_{32} & W_{33} & W_{34} \\ W_{41} & W_{42} & W_{43} & W_{44} \end{bmatrix} \begin{bmatrix} x_1 \\ x_2 \\ x_3 \\ x_4 \end{bmatrix}
    $$

\end{frame}

\section{Convolutional neural networks}
\subsection{Convolutional layers}

% Written by Codex (2026-10-13): convolutional-layers review candidate.
\begin{whiteframe}{Local neighborhoods}

    \begin{columns}[c,onlytextwidth]
        \begin{column}{0.54\textwidth}
            With a \myalert{kernel size} $s=2k+1$, we take the $s \times s$
            neighborhood centered at $(i,j)$, including all $c$ channels.

            We write this local tensor as:
            \[
                P_k(i,j) \sim (s,s,c).
            \]
        \end{column}
        \begin{column}{0.42\textwidth}
            \centering
            \includegraphics[width=\linewidth,height=0.46\textheight,keepaspectratio]{images/cnn/book/patch.pdf}
        \end{column}
    \end{columns}

    \vspace{0.5em}
    \begin{tcolorbox}
        A layer is \myalert{local} if $\idx{H}{i,j,:}$ depends only on $P_k(i,j)$.
    \end{tcolorbox}

\end{whiteframe}

\begin{frame}{Convolutional layer}

    We apply the same affine map and activation to every neighborhood:
    \[
        \idx{H}{i,j,:}
        = \phi\!\left(
        {\color{darkgreen}\vc{W}^{\top}}\,
        \shape{\operatorname{vect}(P_k(i,j))}{(s^2c)}
        + {\color{darkcerulean}\vc{b}}
        \right).
    \]

    We share the weights $\vc{W} \sim (s^2c,c')$ and bias $\vc{b} \sim (c')$
    \textbf{across all locations}.

    This is a \myalert{convolutional layer}. It has $s^2cc' + c'$ parameters,
    regardless of the image height and width.

    The convolution is the affine map inside the parentheses. We apply
    the usual pointwise activation $\phi$ to its output.

\end{frame}

\begin{whiteframe}{Filters and output channels}

    Each column of $\vc{W}$ defines a \myalert{filter}, producing one output
    channel. A spatial slice $\idx{H}{:,:,a}$ is an \myalert{activation map}.

    \begin{figure}
        \centering
        \includegraphics[width=0.7\columnwidth]{images/cnn/source/Convolutive_layer.pdf}
    \end{figure}

\end{whiteframe}

\begin{frame}{Zero-padding}
    \setlength{\parskip}{3pt}
    \setlength{\abovedisplayskip}{9pt}
    \setlength{\belowdisplayskip}{9pt}

    For our four-element sequence, a size-$3$ kernel produces only two outputs
    without padding. We add one zero at each end to obtain four outputs:
    \[
        \widetilde{\vc{x}}
        = \begin{bmatrix}
            {\color{darkcerulean}0} & x_1 & x_2 & x_3 & x_4 & {\color{darkcerulean}0}
        \end{bmatrix}^{\top}.
    \]

    With \myalert{zero padding}, we treat values outside the input as zero.

    \begin{tcolorbox}
        For an odd kernel size $s$, we add $k=(s-1)/2$ zeros at each end to
        preserve the input length when evaluating every position.
    \end{tcolorbox}

    In 2D, we pad each border with $k$ rows or columns: one for a $3 \times 3$
    kernel, two for a $5 \times 5$ kernel.

\end{frame}

\begin{frame}{Running example}
    \setlength{\arraycolsep}{7pt}

    We apply a size-$3$ kernel to the padded sequence, without bias or activation:
    \[
        \begin{bmatrix} h_1 \\ h_2 \\ h_3 \\ h_4 \end{bmatrix}
        =
        \begin{bmatrix}
            {\color{darkgreen}W_1} & {\color{darkgreen}W_2} & {\color{darkgreen}W_3} & 0                      & 0                      & 0                      \\
            0                      & {\color{darkgreen}W_1} & {\color{darkgreen}W_2} & {\color{darkgreen}W_3} & 0                      & 0                      \\
            0                      & 0                      & {\color{darkgreen}W_1} & {\color{darkgreen}W_2} & {\color{darkgreen}W_3} & 0                      \\
            0                      & 0                      & 0                      & {\color{darkgreen}W_1} & {\color{darkgreen}W_2} & {\color{darkgreen}W_3}
        \end{bmatrix}
        \begin{bmatrix}
            {\color{darkcerulean}0} \\ x_1 \\ x_2 \\ x_3 \\ x_4 \\ {\color{darkcerulean}0}
        \end{bmatrix}.
        \label{eq:convolution_example}
    \]

    This is a \myalert{Toeplitz} matrix: the weights are constant along each
    diagonal. We learn three coefficients, compared with sixteen for the
    earlier dense layer.

\end{frame}

\begin{frame}{Translation equivariance}

    Shared weights give the same response to the same neighborhood,
    wherever it appears in the image.

    We write $T_{\delta}$ for a shift by $\delta$ pixels.
    \begin{tcolorbox}
        A layer $f$ is \myalert{translation equivariant} if shifting the
        input shifts its output by the same amount:
        \[
            f(T_{\delta}X)=T_{\delta}f(X).
        \]
    \end{tcolorbox}

    For convolution evaluated at every pixel, this holds away from image
    boundaries. Zero padding can change the response near an edge.

\end{frame}

\begin{frame}{Receptive field}
    \setlength{\parskip}{3pt}

    We combine information across larger regions by stacking local layers.

    \begin{tcolorbox}
        The \myalert{receptive field} of $\idx{H}{i,j,:}$ is the region
        of the input that can affect that output.
    \end{tcolorbox}

    Away from the boundaries, $L$ layers with kernel size $s$, evaluated
    at every pixel, give a receptive field of width:
    \[
        R_L=1+L(s-1).
    \]

    With $3 \times 3$ kernels, the receptive field grows from $3 \times 3$
    to $5 \times 5$ to $7 \times 7$ over the first three layers.

\end{frame}

\begin{frame}[fragile]{Implementation}
    \setlength{\parskip}{3pt}
    \setlength{\abovedisplayskip}{7pt}
    \setlength{\belowdisplayskip}{7pt}

    We can collect flattened, padded neighborhoods as columns of $U(X)$:
    \[
        \shape{\widetilde{Z}}{(c',hw)}
        = \shape{\vc{W}^{\top}}{(c',s^2c)}\,
        \shape{U(X)}{(s^2c,hw)}
        + \vc{b}.
    \]

    \begin{python}[gobble=4,aboveskip=5pt,belowskip=5pt]
        import torch
        X = torch.randn(1, 3, 64, 64)  # (B, C, H, W)
        conv = torch.nn.Conv2d(3, 100, 5, padding=2)
        Z = conv(X)  # (1, 100, 64, 64)
    \end{python}

    In practice, \verb+nn.Conv2d+ uses a specialized implementation for the
    hardware, so we do not construct $U(X)$ explicitly. We will see more in the lab.

    \SlideSource{\href{https://docs.pytorch.org/docs/stable/generated/torch.nn.Unfold.html}{PyTorch: Unfold}
        and \href{https://docs.nvidia.com/deeplearning/performance/dl-performance-convolutional/index.html}{NVIDIA: convolution implementations}.}

\end{frame}

\subsection{Defining the network}

\begin{frame}{Reducing dimensionality}

    We can define a convolutional block by composing convolutional layers $f_i$, each including its activation function:
    %
    $$
        H = (f_l \circ \ldots \circ f_2 \circ f_1)(X)
    $$

    With the padding introduced earlier, convolutional blocks can modify the number of channels, but they keep the spatial resolution $(h,w)$ constant. We might want to reduce the resolution in-between blocks, to make the networks faster and more efficient.

    This is also justified from a signal processing perspective, where \myalert{multi-resolution} filter banks are common.

\end{frame}

\begin{frame}[fragile]{Stride}

    In a convolution with \myalert{stride}, we evaluate every $r$-th position along each spatial axis, where $r$ is the stride parameter.

    For example, for $r=2$, using zero-based indices, we have:
    %
    \[
        \idx{H}{i,j,:} = \phi(\vc{W}^{\top}\operatorname{vect}(P_k({\color{darkred}2i,2j}))+\vc{b}) \,.
    \]
    %
    With the padding introduced earlier, $H \sim (\lceil h/2\rceil,\lceil w/2\rceil,c')$, which becomes $(h/2,w/2,c')$ when $h$ and $w$ are even.

    The \verb+nn.Conv2d+ layer we saw before has a \verb+stride+ parameter.

\end{frame}

\begin{frame}{Max-pooling}

    Alternatively, a \myalert{max-pooling} (or an \myalert{average-pooling}) layer can be used. It computes the maximum (or the average) from small blocks of the input tensor.

    Differently from convolutional layers, it is common to consider even-dimensional blocks (2x2, 4x4, ...). It acts on each channel separately.

\end{frame}

\begin{frame}{Complete architecture of a CNN (1)}

    A standard CNN for classification is then composed by:
    %
    \begin{itemize}
        \item Interleaving convolutional and pooling layers;
        \item Flattening (or \myalert{global pooling});
        \item A classification block.
    \end{itemize}

    Note: with global pooling, the final layer is roughly \myalert{invariant} to a translation. Image boundaries and downsampling can break exact invariance.

    More recent CNNs add many variations on this basic architecture. How to choose the sequence of layers and their hyper-parameters is still an open \textit{model selection} research issue.

\end{frame}

\begin{frame}{Complete architecture of a CNN (2)}
    \setlength{\arraycolsep}{2pt}

    \begin{equation*}
        \begin{array}{rclr}
            \textrm{Block 1:}  & \shape{H}{(h',w',c')} =    & (f_l \circ \ldots \circ f_2 \circ f_1)(X) & \text{(\color{darkred}convolutional or pooling layers)} \\
            \textrm{Block 2a:} & \shape{\vc{h}}{(h'w'c')} = & \text{vect}(H)                            & \text{(\color{darkgreen}flattening)}                    \\
            \textrm{Block 2b:} & \shape{\vc{h}}{(c')} =     & \frac{1}{h'w'}\sum_{i,j} \idx{H}{i,j,:}   & \text{(\color{darkgreen}global pooling)}                \\
            \textrm{Block 3:}  & \vc{y} =                   & \text{softmax}(g(\vc{h}))                 & \text{(\color{blue}e.g., logistic regression)}
        \end{array}
    \end{equation*}

\end{frame}

\begin{frame}{Spatial mixing and channel mixing}
    \setlength{\parskip}{3pt}
    \setlength{\abovedisplayskip}{6pt}
    \setlength{\belowdisplayskip}{6pt}

    The positionwise MLP we saw earlier is equivalent to a $1\times1$ \myalert{pointwise convolution} (same channel map at every pixel).

    With a \myalert{depthwise convolution}, each channel has instead its own spatial
    filter. For $K\sim(s,s,c)$ and $s=2k+1$, the layer is:
    \[
        \idx{D}{i,j,a}
        = \phi\!\left(
        \sum_{u=-k}^{k}\sum_{v=-k}^{k}
        \idx{K}{u+k,v+k,a}\,\idx{X}{i+u,j+v,a}+b_a
        \right).
    \]

    A depthwise convolution followed by a pointwise convolution is called a
    \myalert{depthwise-separable convolution}: we separate spatial mixing
    from channel mixing.

    We can also process groups of channels independently (\textit{group convolution}).

\end{frame}

\section{Convolutions beyond images}
\subsection{From images to sequences}

\begin{frame}{Temporal sequences}
    \setlength{\parskip}{3pt}

    CNNs can be extended easily to other domains having grid-like structure
    of various dimensions.

    For example, consider $n$ steps of a time-series
    $\vc{x}_0,\ldots,\vc{x}_{n-1}\in\mathbb{R}^c$, each step having $c$
    features (e.g., readings from $c$ different sensors).

    We represent it using a matrix:
    \vspace{1.5em}
    \[
        X\sim(\eqnmarkbox[drawred]{seqlength}{n},
        \eqnmarkbox[drawgreen]{seqfeatures}{c}).
    \]
    \annotate[yshift=1em]{above,left}{seqlength}{Length of the sequence}
    \annotate[yshift=1em]{above,right}{seqfeatures}{Features}

    We have a similar format for, e.g., text and DNA sequences.

\end{frame}

\begin{frame}{1D convolutional layers}
    \setlength{\parskip}{4pt}
    \setlength{\abovedisplayskip}{7pt}
    \setlength{\belowdisplayskip}{7pt}

    With kernel size $s=2k+1$, we take the neighborhood $P_k(i)\sim(s,c)$
    around position $i$, including all $c$ features.

    The same definition now gives a \myalert{1D convolutional layer}:
    \[
        \idx{H}{i,:}
        =\phi\!\left(\vc{W}^{\top}\operatorname{vect}(P_k(i))+\vc{b}\right),
        \qquad H\sim(n,c').
    \]

    We share $\vc{W}\sim(sc,c')$ and $\vc{b}\sim(c')$ across positions.
    With stride $1$ and $k$ zeros at each end, the sequence length stays $n$.

    Like in the 2D case, the layer is local and translation equivariant away
    from the boundaries.

\end{frame}

\subsection{Representing text}

\begin{frame}{Handling text data}

    Text must be preprocessed to be handled by a neural network. Commonly:

    \begin{enumerate}
        \item We choose how to split (\myalert{tokenize}) the text, e.g., by word, by character, by byte, by \myalert{subword}, and the maximum number $V$ of unique \myalert{tokens} we can handle (e.g., the $V$ most frequent words). The resulting set of tokens is the \myalert{vocabulary} or \myalert{dictionary}.
        \item Each token in the text is then converted into a discrete ID $z_i \in \{1,\ldots,V\}$, giving us an integer sequence $\vc{z}\sim(n)$.
        \item We \myalert{embed} each token ID into a dense vector \textit{embedding} in $\mathbb{R}^d$ via a trainable matrix $\vc{E}\sim(V,d)$ (embedding matrix).
    \end{enumerate}

    By the end of this procedure we have a matrix $X\sim(n,d)$ of $n$ tokens,
    each represented by $d$ features.

\end{frame}

\begin{frame}{Subword embeddings}
    \setlength{\parskip}{3pt}
    \setlength{\abovedisplayskip}{6pt}
    \setlength{\belowdisplayskip}{6pt}

    Word-level vocabularies can miss unseen words. Character tokenization
    produces longer sequences. With \myalert{subword tokenization}, we reuse
    pieces across words, for example (illustrative):
    \[
        \begin{array}{rcl}
            \texttt{playing} & \longmapsto & (\texttt{play},\texttt{ing}) \\
            \texttt{played}  & \longmapsto & (\texttt{play},\texttt{ed})
        \end{array}
    \]

    Each piece has an ID and a row in $\vc{E}\sim(V,d)$.
    Both words share the embedding for \texttt{play}.

    Unseen words can split into known pieces, and frequent words can stay whole.
    Learned pieces need not match linguistic word parts.

    \SlideSource{Sennrich, Haddow and Birch (2016),
        \href{https://aclanthology.org/P16-1162/}{\textit{Neural Machine Translation of Rare Words with Subword Units}}, ACL.}

\end{frame}

\begin{frame}{Byte pair encoding}
    \setlength{\parskip}{3pt}
    \setlength{\abovedisplayskip}{5pt}
    \setlength{\belowdisplayskip}{5pt}

    We can learn the vocabulary with \myalert{byte pair encoding} (BPE):
    \begin{enumerate}
        \item We start with single characters.
        \item We merge the most frequent adjacent pair into a new token.
        \item We repeat to reach the chosen vocabulary size.
    \end{enumerate}

    Toy corpus: \texttt{hug} $\times5$, \texttt{hugs} $\times3$,
    \texttt{pug} $\times2$. We count pairs within words:
    \[
        \begin{array}{rcll}
            (\texttt{u},\texttt{g})  & \longmapsto & \texttt{ug}  & \quad 10\text{ occurrences} \\
            (\texttt{h},\texttt{ug}) & \longmapsto & \texttt{hug} & \quad 8\text{ occurrences}
        \end{array}
    \]

    On new text, we apply the fixed merge rules in their learned order.
    \myalert{Byte-level BPE} starts from $256$ byte values, so it can represent
    any UTF-8 text.

    \SlideSource{\href{https://huggingface.co/learn/llm-course/en/chapter6/5}{Hugging Face: Byte-Pair Encoding tokenization}.}

\end{frame}

\begin{frame}{Dense embeddings}
    \setlength{\parskip}{4pt}
    \setlength{\abovedisplayskip}{7pt}
    \setlength{\belowdisplayskip}{7pt}

    For embedding we learn an \myalert{embedding matrix} $\vc{E}\sim(V,d)$, with one row for each of the $V$ tokens in the vocabulary. Embedding element $z_i$ of the sequence is just a row lookup:

    \[
        \idx{X}{i,:}=\idx{\vc{E}}{z_i,:}
    \]

    We initialize the embeddings randomly and train them with the network, or take them from some pre-trained model or algorithm.
    For example, GloVe and Word2Vec were two classic algorithms for this purpose.

    \SlideSource{Bolukbasi et al. (2016), \textit{Debiasing word embeddings}, NeurIPS.}

\end{frame}

    \setlength{\abovecaptionskip}{3pt}

    \begin{figure}

        \caption{Using custom embeddings for text classification.}
    \end{figure}

\end{whiteframe}

\begin{frame}{Sequence padding}
    \setlength{\parskip}{4pt}
    \setlength{\abovedisplayskip}{7pt}
    \setlength{\belowdisplayskip}{7pt}

    Two texts may have different lengths. Their embedded matrices
    $X_1\sim(n_1,d)$ and $X_2\sim(n_2,d)$ share the same feature dimension $d$.

    For $n_1>n_2$, we can build a padded mini-batch:
    \[
        X=\operatorname{stack}\!\left(
        X_1,\begin{bmatrix}X_2\\\vc{0}\end{bmatrix}
        \right),
        \qquad X\sim(2,n_1,d).
    \]

    We pad to the maximum sequence length in the mini-batch.
    The added zero rows are placeholders.

    We keep a \myalert{validity mask} of real positions, so padded positions
    do not contribute to pooling or to the loss.

\end{frame}

\subsection{Prediction and generation}

\begin{frame}{Two tasks on a sequence}
    \setlength{\parskip}{4pt}

    From the embedded input $X\sim(n,d)$, the convolutional network produces
    features $H\sim(n,c')$. We can use them in two ways:
    \begin{itemize}
        \item For \myalert{sequence classification}, we pool over the valid
              positions and apply a classifier. We obtain one prediction for the
              whole text, such as its topic or sentiment.
        \item For \myalert{next-token prediction}, we keep the positions and
              apply the same vocabulary classifier at each one:
              \[
                  \vc{p}_i=\operatorname{softmax}(g(\idx{H}{i,:}))\sim(V).
              \]
    \end{itemize}

    At position $i$, $\vc{p}_i$ predicts token $z_{i+1}$. Which input positions
    can we let it use?

\end{frame}

\begin{frame}{Masked convolutions}
    \setlength{\parskip}{1pt}
    \setlength{\abovedisplayskip}{2pt}
    \setlength{\belowdisplayskip}{1.5pt}

    \vspace*{0.5em}
    \begin{tcolorbox}
        For a temporal sequence $X\sim(n,c)$, a \myalert{causal model}
        $H=f(X)$, with $H\sim(n,c')$, is such that $\idx{H}{i,:}$ depends
        only on $\idx{X}{j,:}$ for $j\le i$.
    \end{tcolorbox}

    \[
        \idx{H}{i,:}=\phi\!\left(
        \eqnmarkbox[drawred]{causalweights}{(\vc{W}\odot\vc{M})^{\top}}
        \operatorname{vect}(P_k(i))+\vc{b}\right).
    \]
    \annotate[yshift=-0.4em]{below,right}{causalweights}{Masked weight matrix}

    We flatten $P_k(i)$ row by row. For offset $u$, input feature $a$ and
    output channel $q$, the fixed mask $\vc{M}\sim(sc,c')$ is:
    \[
        \idx{\vc{M}}{(u+k)c+a,q}
        =\begin{cases}1,&u\le0,\\0,&u>0,\end{cases}
        \qquad u\in\{-k,\ldots,k\}.
    \]

\end{frame}

    In a convolution, we mask the weights that would read future positions.
    The prediction of $z_{i+1}$ must not already depend on that token.

\end{whiteframe}

\begin{frame}{Training a next-token predictor}
    \setlength{\parskip}{3pt}
    \setlength{\abovedisplayskip}{6pt}
    \setlength{\belowdisplayskip}{6pt}

    We train on a \myalert{shifted target}. For our three-token text:
    \[
        \begin{array}{rcl}
            \text{Input:}  & \texttt{Check}\quad\texttt{this} & (2,16) \\
            \text{Target:} & \texttt{this}\quad\texttt{out}   & (16,3)
        \end{array}
    \]

    For $n$ valid input positions, we apply the cross-entropy loss at every valid
    position:
    \[
        \mathcal{L}=-\frac{1}{n}\sum_{i=0}^{n-1}
        \log\idx{\vc{p}_i}{z_{i+1}}.
    \]

    During training, we provide the true input tokens (\myalert{teacher forcing}).
    With a causal CNN, all these predictions can be computed in parallel,
    without leaking the future targets.

\end{frame}

\begin{frame}{Autoregressive models}
    \setlength{\parskip}{4pt}
    \setlength{\abovedisplayskip}{7pt}
    \setlength{\belowdisplayskip}{7pt}

    The same next-token predictor can generate text in an
    \myalert{autoregressive} way.

    We prompt it with the beginning of the sequence and sample the next token
    from the predicted distribution. We concatenate its ID to the input and
    call the model again, continuing one token at a time. For example:
    \[
        \texttt{Check}
        \quad\longrightarrow\quad
        \texttt{Check this}
        \quad\longrightarrow\quad
        \texttt{Check this out}\quad\ldots
    \]

    The input now includes the model's own sampled tokens.

    Training can compute many next-token predictions in parallel. Generation
    must wait for each sampled token before predicting the following one.

\end{frame}
