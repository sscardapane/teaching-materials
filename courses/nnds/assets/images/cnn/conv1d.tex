% Editable 1D convolution figure: a sequence X ~ (T,C) drawn as a matrix
% (time along the rows), the size-3 window around one position, and the
% output row it produces in Y ~ (T,C'). Written by Claude (2026-10-30) at
% Simone's request ("the input data is 2D").
\begingroup
\begin{tikzpicture}[x=1cm,y=1cm,
        mcell/.style={draw=drawgray!70,line width=0.4pt,minimum size=0.4cm,
            inner sep=0pt},
        edge/.style={-{Stealth[length=2mm]},line width=0.8pt,draw=drawgray},
        lab/.style={font=\small},
        note/.style={font=\footnotesize,text=drawgray}]
    \def\c{0.4}
    % Padded input: rows -1 and 7 are zeros, rows 0..6 are the sequence.
    \foreach \t in {-1,...,7}{
        \foreach \f in {0,...,3}{
            \ifnum\t<0
                \node[mcell,dashed,fill=white] at (\c*\f,-\c*\t) {};
            \else\ifnum\t>6
                \node[mcell,dashed,fill=white] at (\c*\f,-\c*\t) {};
            \else
                \node[mcell,fill=drawblue_l] at (\c*\f,-\c*\t) {};
            \fi\fi
        }
        \ifnum\t<0 \node[note,anchor=east] at (-0.3,-\c*\t) {$0$};
        \else\ifnum\t>6 \node[note,anchor=east] at (-0.3,-\c*\t) {$0$};
        \else \node[note,anchor=east] at (-0.3,-\c*\t) {$t=\t$};
        \fi\fi
    }
    \draw[drawblue,line width=1pt] (-\c/2,\c/2) rectangle (3.5*\c,-6.5*\c);
    % Window around t = 3.
    \draw[draworange,line width=1.6pt,rounded corners=1pt]
        (-\c/2-0.05,-1.5*\c+0.05) rectangle (3.5*\c+0.05,-4.5*\c-0.05);
    \node[lab,anchor=south] at (1.5*\c,1.25) {$\vc{X}\sim(T,C)$};
    \draw[decorate,decoration={brace,amplitude=4pt},drawgray]
        (-\c/2,0.7) -- (3.5*\c,0.7) node[midway,above=3pt,note] {$C$};

    % The layer, on the row of the window (t = 3).
    \node[lab,draw=drawblue,fill=drawblue_l!25!white,rounded corners=2pt,
        inner sep=5pt] (f) at (4.0,-3*\c) {%
        $\phi\!\left(\vc{W}^{\top}\operatorname{vect}(\vc{P}_k(t))+\vc{b}\right)$};
    \draw[edge] (3.5*\c+0.1,-3*\c) -- (f.west);
    \node[note,anchor=north] at (1.5*\c,-7.5*\c-0.1)
        {window $\vc{P}_k(3)\sim(s,C)$};

    % Output.
    \begin{scope}[shift={(6.6,0)}]
        \foreach \t in {0,...,6}{
            \foreach \f in {0,...,2}{
                \node[mcell,fill=drawgreen_l] at (\c*\f,-\c*\t) {};
            }
        }
        \draw[drawgreen,line width=1pt] (-\c/2,\c/2) rectangle (2.5*\c,-6.5*\c);
        \draw[draworange,line width=1.6pt,rounded corners=1pt]
            (-\c/2-0.05,-2.5*\c+0.05) rectangle (2.5*\c+0.05,-3.5*\c-0.05);
        \node[lab,anchor=south] at (\c,1.25) {$\vc{Y}\sim(T,C')$};
        \draw[decorate,decoration={brace,amplitude=4pt},drawgray]
            (-\c/2,0.7) -- (2.5*\c,0.7) node[midway,above=3pt,note] {$C'$};
    \end{scope}
    \draw[edge] (f.east) -- (6.6-\c/2-0.1,-3*\c);
    \node[note,anchor=north,align=center] at ([yshift=-0.3cm]f.south)
        {same $\vc{W}$ and $\vc{b}$\\at every position $t$};
\end{tikzpicture}
\endgroup
