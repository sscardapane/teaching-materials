% Editable CNN comparison: one pixel's channels versus the complete image.
\begingroup
\newcommand{\cnnImageGrid}[3]{%
    \begin{scope}[shift={(#1,#2)}]
        \foreach \row in {0,1,2}{
            \foreach \col in {0,1,2}{
                \pgfmathtruncatemacro{\chosen}{(\row==1)&&(\col==1)}
                \ifnum#3=2
                    \fill[drawblue_l!65!white] (0.6*\col,0.6*\row)
                        rectangle ++(0.6,0.6);
                \else
                    \fill[drawgray_l] (0.6*\col,0.6*\row)
                        rectangle ++(0.6,0.6);
                \fi
                \draw[drawgray!55,line width=0.4pt] (0.6*\col,0.6*\row)
                    rectangle ++(0.6,0.6);
                \ifnum\chosen=1
                    \ifnum#3=3
                        \fill[drawgreen_l] (0.6*\col,0.6*\row)
                            rectangle ++(0.6,0.6);
                        \draw[drawgreen,line width=1pt] (0.6*\col,0.6*\row)
                            rectangle ++(0.6,0.6);
                    \else
                        \ifnum#3=1
                            \fill[drawblue_l] (0.6*\col,0.6*\row)
                                rectangle ++(0.6,0.6);
                            \draw[drawblue,line width=1pt] (0.6*\col,0.6*\row)
                                rectangle ++(0.6,0.6);
                        \fi
                    \fi
                \fi
            }
        }
        \ifnum#3=2\else
            \node[font=\footnotesize] at (0.9,0.9) {$i,j$};
        \fi
    \end{scope}
}
