% Editable CNN comparison: which input values one output pixel depends on,
% for a positionwise MLP and for a fully-connected layer on the vectorized image.
% Redrawn by Claude (2026-10-15) after Simone's review ("too basic").
% Redrawn by Claude (2026-10-29): channels are separate layers, so the
% positionwise MLP reads a column through all channels and the vectorized
% layer visibly reads every channel (the old stack looked channelwise).
\begingroup
% A 3x3 image with #3 channels, drawn as a box: the front face is the
% H x W grid and the channels go into the depth. #1,#2: lower-left corner;
% #4: 1 = the column of the top-right pixel through all channels, 2 = all
% values; #5: fill colour of the highlighted cells; #6: border colour.
\newcommand{\cnnBox}[6]{%
    \begin{scope}[shift={(#1,#2)}]
        \def\cs{0.4}\def\dx{0.2}\def\dy{0.17}
        % front face
        \foreach \row in {0,1,2}{
            \foreach \col in {0,1,2}{
                \pgfmathtruncatemacro{\on}{(#4==2)||((\row==2)&&(\col==2))}
                \ifnum\on=1 \def\fc{#5}\else\def\fc{white}\fi
                \filldraw[fill=\fc,draw=drawgray!70,line width=0.4pt]
                    (\cs*\col,\cs*\row) rectangle ++(\cs,\cs);
            }
        }
        % top face: one strip of cells per channel
        \foreach \k in {1,...,#3}{
            \foreach \col in {0,1,2}{
                \pgfmathtruncatemacro{\on}{(#4==2)||(\col==2)}
                \ifnum\on=1 \def\fc{#5}\else\def\fc{white}\fi
                \filldraw[fill=\fc,draw=drawgray!70,line width=0.4pt]
                    ({\cs*\col+\dx*(\k-1)},{3*\cs+\dy*(\k-1)}) -- ++(\cs,0)
                    -- ++(\dx,\dy) -- ++(-\cs,0) -- cycle;
            }
        }
        % right face
        \foreach \k in {1,...,#3}{
            \foreach \row in {0,1,2}{
                \pgfmathtruncatemacro{\on}{(#4==2)||(\row==2)}
                \ifnum\on=1 \def\fc{#5}\else\def\fc{white}\fi
                \filldraw[fill=\fc,draw=drawgray!70,line width=0.4pt]
                    ({3*\cs+\dx*(\k-1)},{\cs*\row+\dy*(\k-1)}) -- ++(\dx,\dy)
                    -- ++(0,\cs) -- ++(-\dx,-\dy) -- cycle;
            }
        }
        % outline of the highlighted region
        \ifnum#4=1
            \draw[#6,line width=1pt] (2*\cs,2*\cs) -- (3*\cs,2*\cs)
                -- ++({\dx*#3},{\dy*#3}) -- ++(0,\cs) -- ++(-\cs,0)
                -- ++({-\dx*#3},{-\dy*#3}) -- cycle (3*\cs,3*\cs) -- (3*\cs,2*\cs)
                (3*\cs,3*\cs) -- ++({\dx*#3},{\dy*#3});
        \else
            \draw[#6,line width=1pt] (0,0) -- (3*\cs,0) -- ++({\dx*#3},{\dy*#3})
                -- ++(0,3*\cs) -- ++(-3*\cs,0) -- ++({-\dx*#3},{-\dy*#3}) -- cycle
                (3*\cs,0) -- (3*\cs,3*\cs) -- (0,3*\cs) (3*\cs,3*\cs) -- ++({\dx*#3},{\dy*#3});
        \fi
    \end{scope}
}
